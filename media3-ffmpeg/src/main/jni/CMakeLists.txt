cmake_minimum_required(VERSION 3.21)
project(reelix_ffmpeg C CXX)
set(CMAKE_CXX_STANDARD 11)
foreach(lib avutil swresample avcodec)
    add_library(${lib} STATIC IMPORTED)
    set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${FFMPEG_ROOT}/lib/lib${lib}.a")
endforeach()
add_library(ffmpegJNI SHARED ffmpeg_jni.cc)
target_include_directories(ffmpegJNI PRIVATE "${FFMPEG_ROOT}/include")
target_link_libraries(ffmpegJNI PRIVATE swresample avcodec avutil android log m z)
target_link_options(ffmpegJNI PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-Bsymbolic")
